# The library itself: four QML modules, one C++ core.
#
#   QGravityUI.Core      enums, input-mode tracking, theme scope, clipboard
#   QGravityUI.Tokens    generated colours, typography, component metrics
#   QGravityUI.Icons     the icon subset, packed by tools/gen-icons
#   QGravityUI.Controls   the 67 controls
#
# The build order is the dependency order; each module declares the ones it
# imports so qmlimportscanner can follow them.
add_subdirectory(Core)
add_subdirectory(Tokens)
add_subdirectory(Icons)
add_subdirectory(Controls)

# One target to link instead of four. The names below are what the generated
# qmldir files name in their `linkTarget` lines, so they stay as they are --
# see cmake/Install.cmake for why that matters -- and this is a plain
# INTERFACE facade over them.
#
#   add_subdirectory(third_party/QGravityUI)
#   target_link_libraries(app PRIVATE QGravityUI)
add_library(QGravityUI INTERFACE)

set(QGRAVITYUI_MODULES core tokens icons controls)

foreach(_gq_module IN LISTS QGRAVITYUI_MODULES)
    target_link_libraries(QGravityUI INTERFACE qgravityui_${_gq_module})

    # These modules are static, so the QML plugin does not load itself: the
    # plugin library has to be linked and its Q_IMPORT_PLUGIN object
    # (<plugin>_init, generated by qt_add_qml_module) has to come with it.
    # Qt normally arranges this from what qmlimportscanner finds -- but the
    # scanner only looks along the import path, and a submodule's modules are
    # not on the consumer's. Linking them here is what makes
    # `target_link_libraries(app PRIVATE QGravityUI)` sufficient on its own,
    # with no QT_QML_IMPORT_PATH to set and no IMPORTS to repeat. Without it
    # the consumer builds and links fine and then fails at run time with
    # `module "QGravityUI.Core" plugin "qgravityui_coreplugin" not found`.
    target_link_libraries(QGravityUI INTERFACE qgravityui_${_gq_module}plugin)
    if(TARGET qgravityui_${_gq_module}plugin_init)
        target_link_libraries(QGravityUI INTERFACE qgravityui_${_gq_module}plugin_init)
    endif()
endforeach()
